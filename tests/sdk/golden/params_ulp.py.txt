# ── GENIE Cloud SDK 脚本 · 由 GENIE Studio 表单自动生成 ──
# 本地运行: pip install genie-sdk  # from <YOUR_GENIE_RELEASE_CHANNEL>
import genie

f = genie.connect('https://genie.example.com', api_key="YOUR_API_KEY", username="YOUR_USERNAME")

# parameters
d = 50*1e-9  # 原文 "50[nm]"
w = 50*1e-9  # 原文 "50*1e-9"
half = w/2  # 原文 "w/2"
plain = 5e-8  # 原文 5e-8

# ── Domain 域 ──
f.domain.x_range = (-d, d)
f.domain.y_range = (-w, w)
f.domain.z_range = (-1e-6, 1e-6)
f.domain.total_time = 6e-14

# ── Mesh 网格 ──
f.mesh.dx = d/4

# ── Materials 材料 ──
f.mat.add_const() \
    .name('sio2') \
    .eps(2.1)

# ── Geometry 结构 ──
f.geom.add_box().material('sio2') \
    .x_range(-half, half) \
    .y_range(-5e-7, 5e-7) \
    .z_range(-1e-7, 1e-7)

# ── Source 源 ──
f.src.add_soft_source() \
    .freq(3e14) \
    .bandwidth(6e13) \
    .name('s0')

# ── Monitor 监视器 ──
f.mon.add_time() \
    .name('pt') \
    .field('Ex')

# ── Run 运行 ──
f.run()

# ── 结果（纯数据 numpy） ──
t_pt = f.get_time('pt')        # t_pt.time / t_pt.fields['Ey']
print(t_pt.time[-1])        # 末时刻 (s)
