# ── Domain 域 ──
f.domain.x_range = (-1e-6, 1e-6)

# ── Mesh 网格 ──
f.mesh.dy = 3e-8

# ── Materials 材料 ──
f.mat.add_const() \
    .eps(2.1)

# ── Geometry 结构 ──
f.geom.add_box().material('sio2') \
    .x_range(-1e-7, 1e-7)

# ── Monitor 监视器 ──
f.mon.add_time() \
    .center(0, 0, 2e-7)

# ── Run 运行 ──
f.run()

# ── 结果（纯数据 numpy） ──
t_m = f.get_time('')        # t_m.time / t_m.fields['Ey']
print(t_m.time[-1])        # 末时刻 (s)
